/* ------------------------------ MAESTRO CSV - CORREGIDO (maneja tabs, espacios y formato científico) --------------------------------*/ document.getElementById('fileInput').addEventListener('change', function(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(evt) { const text = evt.target.result.replace(/\r/g,''); const lines = text.split('\n').filter(l => l.trim() !== ''); maestro = {}; let importados = 0; let errores = 0; for (let i = 1; i < lines.length; i++) { let linea = lines[i]; // Reemplazar tabuladores por comas linea = linea.replace(/\t/g, ','); // Limpiar espacios extra y múltiples comas linea = linea.replace(/\s*,\s*/g, ','); linea = linea.replace(/,+/g, ','); // Eliminar espacios al inicio y final de la línea linea = linea.trim(); // Si la línea está vacía o solo tiene comas, saltar if (!linea || linea.replace(/,/g, '').trim() === '') continue; const cols = linea.split(',').map(c => c.trim()); // Asegurar que tenemos al menos 4 columnas if (cols.length >= 4) { const idArticulo = cols[0] ? cols[0].trim() : ''; const nombreArticulo = cols[1] ? cols[1].trim() : ''; let codigoBarras = cols[2] ? cols[2].trim() : ''; const stockActual = cols[3] ? cols[3].trim() : '0'; // Si algún campo obligatorio está vacío, saltar if (!idArticulo || !nombreArticulo || !codigoBarras) { errores++; continue; } // CONVERTIR NÚMERO CIENTÍFICO A TEXTO COMPLETO if (codigoBarras && typeof codigoBarras === 'string') { // Eliminar puntos de miles codigoBarras = codigoBarras.replace(/\./g, ''); // Si tiene formato científico como "8,42136E+12" o "8.42136E+12" if (codigoBarras.includes('E') || codigoBarras.includes('e')) { const numStr = codigoBarras.replace(',', '.'); const num = parseFloat(numStr); if (!isNaN(num)) { codigoBarras = num.toFixed(0); } } } // Asegurar que el código de barras es string codigoBarras = String(codigoBarras); if (codigoBarras && codigoBarras !== '' && codigoBarras !== '0') { maestro[codigoBarras] = { idArticulo: String(idArticulo), nombreArticulo: String(nombreArticulo), stockActual: String(stockActual || '0') }; importados++; } else { errores++; } } else { errores++; } } alert(`✅ Maestro cargado correctamente.\n📦 Productos importados: ${importados}\n⚠️ Errores/omitidos: ${errores}\n📊 Total en maestro: ${Object.keys(maestro).length}`); }; reader.readAsText(file); });
